\chapter{Higher differentials and rules of calculus}
\label{chap:higher}

There are two natural ways to say that a map $f \colon U \to F$ is $\Ck{k}$. The one we adopted
in Definition~\ref{def:ck} is recursive: $f$ is $\Ck{k+1}$ when $\dd f$, regarded as a map on the
open set $U \times E$, is $\Ck{k}$. This is the form used by Glöckner, and it is the convenient
one for proofs by induction. The classical definition, due to Bastiani \cite{Bas64} in 1964,
asks instead that the iterated directional derivatives $\dd^{(j)} f(x; y_1, \dots, y_j)$ exist
and be jointly continuous in the point and in all the directions, for $j \leq k$.

In this chapter we prove that both definitions agree, and we derive from this the basic
structure of the higher differentials: if $f$ is $\Ck{k+r}$, then $\dd^{(k)} f$ is again
$\Ck{r}$, and for each $x$ the map $\dd^{(k)} f(x; \cdot)$ is continuous, symmetric and
$k$-linear. We first define the iterated differentials and Bastiani's notion, and prove a
formula (the slice lemma) relating the iterated differentials of $f$ to those of $\dd f$; this
formula is the bridge between the two definitions. We then deal with regularity and symmetry, and
the last section collects the rules of calculus used in practice: the product rule, Taylor's
formula and the second-order chain rule. Our reference is \cite[\S1.3]{GN26}.

\section{Iterated differentials}

\begin{definition}[Iterated differentials]
  \label{def:iterated}
  \lean{LocallyConvex.Analysis.iteratedDMB}
  \leanok
  \uses{def:dir-deriv}
  For $f \colon E \to F$, $x \in E$ and $y_1, \dots, y_k \in E$ we define
  \[
    \dd^{(0)} f(x) = f(x), \qquad
    \dd^{(k+1)} f(x; y_1, \dots, y_{k+1})
      = \dd\bigl(z \mapsto \dd^{(k)} f(z; y_1, \dots, y_k)\bigr)(x)(y_{k+1}),
  \]
  that is, $\dd^{(k)} f(x; y_1, \dots, y_k) = (D_{y_k} \cdots D_{y_1} f)(x)$.
\end{definition}

So $\dd^{(k)} f(x; y_1, \dots, y_k)$ is obtained by differentiating $f$ successively in the
directions $y_1, \dots, y_k$, keeping the directions fixed and moving only the point. Note that
$\dd^{(1)} f(x; y) = \dd f(x)(y)$.

\begin{definition}[Bastiani's $\Ck{k}$ maps]
  \label{def:bastiani}
  \lean{LocallyConvex.Analysis.HasContIterDerivOn}
  \leanok
  \uses{def:iterated}
  A map $f$ has continuous iterated derivatives of order $k$ on $U$ if, for every $j \leq k$,
  the iterated differentials $\dd^{(j)} f(x; y_1, \dots, y_j)$ exist for $x \in U$ and are
  continuous on $U \times E^j$.
\end{definition}

\begin{remark}
  In a Banach space the $k$-th Fréchet derivative is a map from $U$ into the space of
  continuous $k$-linear maps $E^k \to F$, and its continuity is measured in the operator norm.
  Here no topology is put on spaces of multilinear maps: what is required is the joint
  continuity of $\dd^{(j)} f$ on $U \times E^j$. This is the same choice already made for
  $k = 1$ in Definition~\ref{def:ck}, and it is what allows the theory to work without norms.
\end{remark}

The bridge between the two definitions is a formula that expresses the iterated differentials
of $f$ through those of $\dd f$, in directions of the form $(y, 0)$.

\begin{lemma}[Slices]
  \label{lem:slice}
  \lean{LocallyConvex.Analysis.iteratedDMB_slice, LocallyConvex.Analysis.iteratedDMB_succ_eq_dMB}
  \leanok
  \uses{def:iterated}
  For $g \colon E \times E' \to F$ and $a \in E'$,
  \[
    \dd^{(k)}\bigl(g(\cdot, a)\bigr)(x; v_1, \dots, v_k)
      = \dd^{(k)} g\bigl((x, a); (v_1, 0), \dots, (v_k, 0)\bigr).
  \]
  In particular
  \[
    \dd^{(k+1)} f(x; y_0, \dots, y_k)
      = \dd^{(k)}(\dd f)\bigl((x, y_0); (y_1, 0), \dots, (y_k, 0)\bigr).
  \]
\end{lemma}

In words: differentiating the partial map $g(\cdot, a)$ is the same as differentiating $g$ in
directions that do not move the second variable. The second formula is the case $g = \dd f$,
$a = y_0$.

\begin{proof}
  \leanok
  Both sides are limits of the same difference quotients, and the claim follows by induction on
  $k$. The second formula is \cite[(1.36)]{GN26}.
\end{proof}

\begin{theorem}[The two definitions of $\Ck{k}$ agree]
  \label{thm:ck-iff-bastiani}
  \lean{LocallyConvex.Analysis.contDiffMBOn_iff_hasContIterDerivOn}
  \leanok
  \uses{def:ck, def:bastiani, lem:slice, thm:linear, thm:partials}
  A map $f$ is $\Ck{k}$ on $U$ in the sense of Definition~\ref{def:ck} if and only if it has
  continuous iterated derivatives of order $k$ on $U$.
\end{theorem}

From now on we use whichever of the two descriptions is more convenient.

\begin{proof}
  \leanok
  Both implications go by induction on $k$, through Lemma~\ref{lem:slice}. If $f$ is
  $\Ck{k+1}$, the iterated differentials of $f$ are slices of those of $\dd f$, which is
  $\Ck{k}$; hence they exist and are continuous. For the converse, the derivatives of $\dd f$
  in a general direction $(h, w)$ split, by linearity in the second variable and
  Theorem~\ref{thm:partials}, into a derivative in the direction $(h, 0)$, which is an iterated
  differential of $f$, and one in the direction $(0, w)$, which is again of the form $\dd f$.
  The details can be found in \cite[1.3.10 and 1.3.12]{GN26}.
\end{proof}

\section{Regularity and symmetry}

\begin{theorem}[Regularity of the differentials]
  \label{thm:iterated-regular}
  \lean{LocallyConvex.Analysis.ContDiffMBOn.contDiffMBOn_iteratedDMB}
  \leanok
  \uses{thm:ck-iff-bastiani}
  If $f$ is $\Ck{k + r}$ on $U$, then $\dd^{(k)} f$ is $\Ck{r}$ on $U \times E^k$.
\end{theorem}

Thus each differentiation costs exactly one order of regularity, as in the Banach case; in
particular, the iterated differentials of a smooth map are smooth.

\begin{proof}
  \leanok
  We argue by induction on $k$, using that $\dd^{(k+1)} f$ is, up to reordering the variables,
  the derivative of $\dd^{(k)} f$ in directions of the form $(y, 0)$. This is \cite[1.3.13]{GN26}.
\end{proof}

\begin{theorem}[Schwarz]
  \label{thm:schwarz}
  \lean{LocallyConvex.Analysis.ContDiffMBOn.iteratedDMB_perm, LocallyConvex.Analysis.ContDiffMBOn.exists_iteratedDMB_multilinear}
  \leanok
  \uses{thm:symm2, thm:linear, thm:iterated-regular}
  If $f$ is $\Ck{k}$ on $U$ and $x \in U$, then $\dd^{(k)} f(x; \cdot)$ is a continuous
  symmetric $k$-linear map $E^k \to F$.
\end{theorem}

This is what allows us to regard $\dd^{(k)} f(x)$ as the $k$-th differential of $f$ at $x$, in
the same sense as in finite dimension: the order in which we differentiate is irrelevant, and
the result depends linearly on each direction.

\begin{proof}
  \leanok
  We begin with symmetry. It suffices to consider transpositions of adjacent variables, and
  these reduce to Theorem~\ref{thm:symm2} applied to an iterated differential of lower order.
  Linearity in the last variable is Theorem~\ref{thm:linear} applied to $\dd^{(k-1)} f$; by
  symmetry, the map is linear in every variable. Continuity follows from the joint continuity
  of $\dd^{(k)} f$. This is \cite[1.3.17]{GN26}.
\end{proof}

\begin{remark}
  A word of caution: the continuity in Theorem~\ref{thm:schwarz} is continuity of the
  $k$-linear map $\dd^{(k)} f(x; \cdot)$ on $E^k$, for a fixed $x$. Nothing is claimed about
  the dependence of this multilinear map on $x$ in a topology on a space of multilinear maps;
  the regularity in $x$ is expressed by the joint statement of
  Theorem~\ref{thm:iterated-regular}.
\end{remark}


\section{Rules of calculus}
\label{sec:rules}

With the chain rule (Theorem~\ref{thm:chain-rule}) and the structure of the higher
differentials (Chapter~\ref{chap:higher}) at hand, we can now derive the remaining rules of
calculus that are used in practice. All of them are classical in the Banach setting, and their
statements are the familiar ones; what has to be checked is that the proofs survive without
norms. In each case the argument either reduces to the chain rule or, through a continuous
linear functional, to the calculus of real functions of one variable.

We begin with continuous bilinear maps and the product rule. We then compute the derivatives
of a map along a line, which leads to Taylor's formula in a weak form, valid without any
completeness assumption on $F$. The section closes with the chain rule for second
differentials. The references are the results of \cite{GN26} quoted in each proof.

\subsection{Bilinear maps and the product rule}

\begin{theorem}[Bilinear maps]
  \label{thm:bilinear}
  \lean{LocallyConvex.Analysis.hasDirDerivAt_bilin, LocallyConvex.Analysis.contDiffMBOn_bilin}
  \leanok
  \uses{def:ck, thm:partials}
  Let $\beta \colon A \times B \to F$ be continuous and bilinear. Then $\beta$ is $\Ck{k}$ for
  every $k$, with $\dd \beta\bigl((a, b), (h, w)\bigr) = \beta(a, w) + \beta(h, b)$.
\end{theorem}

\begin{proof}
  \leanok
  The difference quotient of $\beta$ at $(a,b)$ in the direction $(h,w)$ is
  $\beta(a, w) + \beta(h, b) + t\,\beta(h, w)$, and letting $t \to 0$ gives the formula. The
  differential is therefore a sum of continuous bilinear maps composed with continuous linear
  maps, and induction on $k$ completes the proof. This is \cite[1.2.3 and 1.3.5]{GN26}.
\end{proof}

Note that continuity of $\beta$ means joint continuity on $A \times B$. This cannot be
weakened to separate continuity: for $b = 0$ and $h = 0$ the differential above reduces to
$\beta(a, w)$, which must be jointly continuous in $(a, w)$.

\begin{corollary}[Product rule]
  \label{cor:product}
  \lean{LocallyConvex.Analysis.ContDiffMBOn.bilin, LocallyConvex.Analysis.ContDiffMBOn.dMB_bilin}
  \leanok
  \uses{thm:bilinear, thm:chain-rule, lem:prod}
  If $f \colon U \to A$ and $g \colon U \to B$ are $\Ck{k}$ and $\beta$ is continuous and
  bilinear, then $x \mapsto \beta(f(x), g(x))$ is $\Ck{k}$ and
  \[
    \dd\bigl(\beta(f, g)\bigr)(x)(h) = \beta\bigl(f(x), \dd g(x)(h)\bigr)
      + \beta\bigl(\dd f(x)(h), g(x)\bigr).
  \]
\end{corollary}

\begin{proof}
  \leanok
  It is immediate from the chain rule applied to $\beta \circ (f, g)$, together with
  Theorem~\ref{thm:bilinear}.
\end{proof}

\begin{example}
  Let $\beta \colon E \times E \to F$ be continuous and bilinear, and let
  $q(x) = \beta(x, x)$. Taking $f = g = \mathrm{id}_E$ in Corollary~\ref{cor:product}, we
  obtain that $q$ is smooth, with $\dd q(x)(h) = \beta(x, h) + \beta(h, x)$. When $\beta$ is
  symmetric this is the familiar $2\beta(x, h)$.
\end{example}

\subsection{Taylor's formula}

To pass from the differentials of $f$ to the values of $f$ we restrict $f$ to a line and
compose with a continuous linear functional. The resulting real function of one variable is
as regular as $f$, and its derivatives are the iterated differentials of $f$ along the line.

\begin{lemma}[Derivatives along a line]
  \label{lem:line}
  \lean{LocallyConvex.Analysis.ContDiffMBOn.hasDerivAt_iteratedDMB_line, LocallyConvex.Analysis.ContDiffMBOn.continuousOn_iteratedDMB_line}
  \leanok
  \uses{thm:iterated-regular}
  Let $f$ be $\Ck{k}$ on $U$, $j < k$ and $\psi$ a continuous linear functional on $F$. Then
  the real function $s \mapsto \psi\bigl(\dd^{(j)} f(x + sh; h, \dots, h)\bigr)$ has derivative
  $\psi\bigl(\dd^{(j+1)} f(x + sh; h, \dots, h)\bigr)$ wherever $x + sh \in U$.
\end{lemma}

\begin{proof}
  \leanok
  By the chain rule along the line $s \mapsto (x + sh, h, \dots, h)$, whose derivative is
  $(h, 0, \dots, 0)$.
\end{proof}

\begin{theorem}[Taylor's formula, weak form]
  \label{thm:taylor}
  \lean{LocallyConvex.Analysis.ContDiffMBOn.taylor}
  \leanok
  \uses{lem:line}
  Let $f$ be $\Ck{k+1}$ on a set containing the segment $[x, x+h]$. For every continuous linear
  functional $\psi$ on $F$,
  \[
    \psi\bigl(f(x + h)\bigr) = \sum_{j=0}^{k} \frac{1}{j!}\,
      \psi\bigl(\dd^{(j)} f(x; h, \dots, h)\bigr)
      + \int_0^1 \frac{(1-t)^k}{k!}\,
      \psi\bigl(\dd^{(k+1)} f(x + th; h, \dots, h)\bigr)\, dt .
  \]
\end{theorem}

\begin{proof}
  \leanok
  We apply Taylor's formula with integral remainder to the real function
  $s \mapsto \psi(f(x + sh))$, whose derivatives are given by Lemma~\ref{lem:line}. Compare
  \cite[1.6.1 and 1.6.14]{GN26}, where the formula is stated for the integrals in $F$; the weak form
  needs no completeness of $F$.
\end{proof}

\begin{remark}
  The remainder in Theorem~\ref{thm:taylor} is a real integral, one for each functional $\psi$.
  Writing it as a vector of $F$ requires the weak integral of
  $t \mapsto \frac{(1-t)^k}{k!}\,\dd^{(k+1)} f(x + th; h, \dots, h)$ to exist, which in general
  depends on a completeness property of $F$ (Chapters~\ref{chap:integrals}
  and~\ref{chap:completeness}). Since continuous linear functionals separate points, the weak
  form determines $f(x + h)$ all the same.
\end{remark}

\subsection{The second-order chain rule}

\begin{theorem}[Second-order chain rule]
  \label{thm:chain-second}
  \lean{LocallyConvex.Analysis.dMB_dMB_comp}
  \leanok
  \uses{thm:chain-rule}
  Let $\tau \colon U \to V$ and $g \colon V \to F$ be $\Ck{2}$, with $U \subseteq E$ and
  $V \subseteq E'$ open. For $y \in U$ and $a, b \in E$,
  \[
    \dd^{(2)} (g \circ \tau)(y; a, b)
      = \dd^{(2)} g\bigl(\tau y; \dd\tau(y)(a), \dd\tau(y)(b)\bigr)
        + \dd g(\tau y)\bigl(\dd^{(2)} \tau(y; a, b)\bigr).
  \]
\end{theorem}

The first term is what one would guess by applying the chain rule twice naively; the second
involves the second differential of $\tau$, and vanishes when $\tau$ is the restriction of a continuous
linear map, since then $\dd\tau(y)(b)$ does not depend on $y$.

\begin{proof}
  \leanok
  On $U \times E$, the first-order chain rule gives $\dd(g \circ \tau) = \dd g \circ \Phi$ with
  $\Phi(y', b') = (\tau y', \dd\tau(y')(b'))$. We differentiate this once more with the
  first-order chain rule. The derivative of $\Phi$ at $(y, b)$ in the direction $(a, 0)$ is
  $(\dd\tau(y)(a), \dd^{(2)}\tau(y; a, b))$. Splitting it as
  $(\dd\tau(y)(a), 0) + (0, \dd^{(2)}\tau(y; a, b))$, the first term gives $\dd^{(2)} g$, and
  the second gives $\dd g$, because $\dd g(\tau y)$ is linear.
\end{proof}

\subsection{Operations on \texorpdfstring{$\Ck{k}$}{Ck} maps}

\begin{proposition}[Operations]
  \label{prop:operations}
  \lean{LocallyConvex.Analysis.hasDirDerivAt_pi, LocallyConvex.Analysis.HasDirDerivAt.sum,
    LocallyConvex.Analysis.HasDirDerivAt.clm_apply, LocallyConvex.Analysis.HasDirDerivAt.smul_fun,
    LocallyConvex.Analysis.contDiffMBOn_pi, LocallyConvex.Analysis.ContDiffMBOn.const_smul,
    LocallyConvex.Analysis.ContDiffMBOn.sub, LocallyConvex.Analysis.ContDiffMBOn.smul_fun,
    LocallyConvex.Analysis.ContDiffMBOn.prodMap, LocallyConvex.Analysis.SmoothMBOn.prodMap,
    LocallyConvex.Analysis.ContDiffMBOn.codRestrict,
    LocallyConvex.Analysis.contDiffMBOn_of_isClosedEmbedding,
    LocallyConvex.Analysis.dMB_eq_comp_of_eventuallyEq, LocallyConvex.Analysis.dMB_comp_affine,
    LocallyConvex.Analysis.SmoothMBOn.dMB_dir, LocallyConvex.Analysis.dMB_eq_fderiv,
    LocallyConvex.Analysis.dMB_dMB_eq_fderiv_fderiv}
  \leanok
  \uses{def:ck, thm:chain-rule, thm:bilinear}
  Directional derivatives and $\Ck{k}$ maps behave as expected under the elementary operations:
  maps into products are differentiated componentwise, and finite sums, multiples, differences,
  products $f \cdot g$ of a real function with a vector-valued one, and products of maps
  $(x, y) \mapsto (g(x), h(y))$ of $\Ck{k}$ maps are $\Ck{k}$. A $\Ck{k}$ map with values in a
  closed subspace is $\Ck{k}$ into the subspace \cite[compare Lemma~1.3.19]{GN26}. The
  differential depends only on the germ of the map, and on normed spaces the first and second
  Michal--Bastiani differentials of $\Ck{1}$ and $\Ck{2}$ maps are the Fréchet derivatives.
\end{proposition}

\begin{proof}
  \leanok
  Each statement reduces to the chain rule and continuous linear maps, or to the product rule
  for the continuous bilinear map $(t, v) \mapsto tv$; the closed subspace case uses that
  difference quotients stay in the subspace.
\end{proof}
